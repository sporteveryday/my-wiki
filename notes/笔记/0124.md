### 1. LeetCode 3 - 无重复字符的最长子串（滑动窗口 + Map/Set）

```javascript
var lengthOfLongestSubstring = function (s) {
  let right = 0;
  let left = 0;
  let max = 0;
  let map = new Map();

  while (right < s.length) {
    if (!map.has(s[right])) {
      // 当前字符没出现过
      map.set(s[right], right); // 记录它的位置
      right++; // 窗口右边界扩张
      max = Math.max(max, right - left); // 更新最长长度
    } else {
      // 发现重复了
      map.delete(s[left]); // 尝试删除左边界的字符
      left++; // 左边界右移
    }
  }
  return max;
};
```

// 时间 O(n)，空间 O(min(n, 字符集大小))

### 2. LeetCode 206 - 反转链表（迭代版，字节最爱）

```javascript
function reverseList(head) {
  let prev = null;
  let curr = head;

  while (curr) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}
```

// 时间 O(n)，空间 O(1)

### 3. LeetCode 146 - LRU Cache（Map + 双向链表）

```javascript
/**
 * @param {number} capacity
 */
var LRUCache = function (capacity) {
  this.map = new Map();
  this.capacity = capacity;
};

/**
 * @param {number} key
 * @return {number}
 */
LRUCache.prototype.get = function (key) {
  if (this.map.has(key)) {
    let value = this.map.get(key);
    this.map.delete(key); // 删除后，再 set ，相当于更新到 map 最后一位
    this.map.set(key, value);
    return value;
  } else {
    return -1;
  }
};

/**
 * @param {number} key
 * @param {number} value
 * @return {void}
 */
LRUCache.prototype.put = function (key, value) {
  // 如果已有，那就要更新，即要先删了再进行后面的 set
  if (this.map.has(key)) {
    this.map.delete(key);
  }
  this.map.set(key, value);
  // put 后判断是否超载
  if (this.map.size > this.capacity) {
    this.map.delete(this.map.keys().next().value);
  }
};
```

// 时间 O(1)，空间 O(capacity)

### 4. LeetCode 1 - 两数之和

```javascript
function twoSum(nums, target) {
  const map = new Map(); // 用来存「已经看过的数 → 它的下标」

  for (let i = 0; i < nums.length; i++) {
    // 一次遍历数组
    const complement = target - nums[i]; // 当前数配对后需要的另一个数

    if (map.has(complement)) {
      // 如果之前已经见过这个需要的数
      return [map.get(complement), i]; // 返回两个数的下标（之前那个 + 当前这个）
    }

    map.set(nums[i], i); // 把当前数和它的下标记录下来
  }

  return []; // 遍历完都没找到，返回空数组
}
```

// 时间 O(n)，空间 O(n)

### 5. LeetCode 415 - 字符串相加（大数加法）

```javascript
function addStrings(num1, num2) {
  let i = num1.length - 1;
  let j = num2.length - 1;
  let carry = 0;
  let result = "";

  while (i >= 0 || j >= 0 || carry) {
    const digit1 = i >= 0 ? num1[i--] - "0" : 0;
    const digit2 = j >= 0 ? num2[j--] - "0" : 0;
    const sum = digit1 + digit2 + carry;
    result = (sum % 10) + result;
    carry = Math.floor(sum / 10);
  }
  return result;
}
```

### 6. LeetCode 20 - 有效的括号

```javascript
function isValid(s) {
  const stack = [];
  const map = { ")": "(", "}": "{", "]": "[" };

  for (const char of s) {
    if (char === "(" || char === "{" || char === "[") {
      stack.push(char);
    } else {
      if (stack.length === 0 || stack.pop() !== map[char]) {
        return false;
      }
    }
  }
  return stack.length === 0;
}
```

### 8. LeetCode 215 - 数组中的第 K 个最大元素（快选 / 堆）

最常用快选 partition 写法：

```javascript
function findKthLargest(nums, k) {
  function quickSelect(left, right, k) {
    if (left === right) return nums[left];

    const pivotIndex = partition(left, right);
    if (k === pivotIndex) return nums[k];
    else if (k < pivotIndex) return quickSelect(left, pivotIndex - 1, k);
    else return quickSelect(pivotIndex + 1, right, k);
  }

  function partition(left, right) {
    const pivot = nums[right];
    let i = left;
    for (let j = left; j < right; j++) {
      if (nums[j] >= pivot) {
        // 降序 → 第k大
        [nums[i], nums[j]] = [nums[j], nums[i]];
        i++;
      }
    }
    [nums[i], nums[right]] = [nums[right], nums[i]];
    return i;
  }

  return quickSelect(0, nums.length - 1, k - 1);
}
```

// 平均 O(n)，最坏 O(n²)

### 9. LeetCode 23 - 合并 K 个有序链表（使用最小堆）

```javascript
class ListNode {
  constructor(val = 0, next = null) {
    this.val = val;
    this.next = next;
  }
}

function mergeKLists(lists) {
  if (!lists || lists.length === 0) return null;

  // 最小堆：用数组模拟 + sort（面试时允许，实际生产用 PriorityQueue）
  const minHeap = [];

  // 把所有非空链表头节点放入堆
  for (let list of lists) {
    if (list) {
      minHeap.push(list);
    }
  }

  // 按节点值排序（最小堆）
  minHeap.sort((a, b) => a.val - b.val);

  const dummy = new ListNode(0);
  let tail = dummy;

  while (minHeap.length > 0) {
    // 取出最小节点
    const node = minHeap.shift();
    tail.next = node;
    tail = tail.next;

    // 如果有下一个节点，加入堆
    if (node.next) {
      minHeap.push(node.next);
      // 重新排序（模拟堆调整）
      minHeap.sort((a, b) => a.val - b.val);
    }
  }

  return dummy.next;
}
```

// 时间：O(N log k)，N 是所有节点总数，k 是链表数  
// 空间：O(k)

### 10. LeetCode 5 - 最长回文子串（中心扩散法，面试首选）

```javascript
function longestPalindrome(s) {
  if (!s || s.length < 1) return "";

  let start = 0;
  let maxLen = 1;

  function expandAroundCenter(left, right) {
    while (left >= 0 && right < s.length && s[left] === s[right]) {
      const currentLen = right - left + 1;
      if (currentLen > maxLen) {
        start = left;
        maxLen = currentLen;
      }
      left--;
      right++;
    }
  }

  for (let i = 0; i < s.length; i++) {
    // 奇数长度回文（以 i 为中心）
    expandAroundCenter(i, i);
    // 偶数长度回文（以 i 和 i+1 为中心）
    expandAroundCenter(i, i + 1);
  }

  return s.slice(start, start + maxLen);
}
```

// 时间 O(n²)，空间 O(1)

### 12. LeetCode 739 - 每日温度（单调栈）

```javascript
function dailyTemperatures(temperatures) {
  const n = temperatures.length;
  const answer = new Array(n).fill(0);
  const stack = []; // 存索引，温度单调递减栈

  for (let curr = 0; curr < n; curr++) {
    while (
      stack.length > 0 &&
      temperatures[stack[stack.length - 1]] < temperatures[curr]
    ) {
      const prev = stack.pop();
      answer[prev] = curr - prev;
    }
    stack.push(curr);
  }

  return answer;
}
```

// 时间 O(n)，空间 O(n)

### 13. LeetCode 438 - 找到字符串中所有字母异位词（滑动窗口）

```javascript
function findAnagrams(s, p) {
  const result = [];
  if (s.length < p.length) return result;

  const pCount = new Array(26).fill(0);
  const sCount = new Array(26).fill(0);

  // 初始化 p 的字符计数
  for (let char of p) {
    pCount[char.charCodeAt(0) - 97]++;
  }

  // 初始化窗口前 p.length 个字符
  for (let i = 0; i < p.length; i++) {
    sCount[s[i].charCodeAt(0) - 97]++;
  }

  if (arraysEqual(pCount, sCount)) {
    result.push(0);
  }

  // 滑动窗口
  for (let i = p.length; i < s.length; i++) {
    // 移除左边字符
    sCount[s[i - p.length].charCodeAt(0) - 97]--;
    // 添加右边字符
    sCount[s[i].charCodeAt(0) - 97]++;

    if (arraysEqual(pCount, sCount)) {
      result.push(i - p.length + 1);
    }
  }

  return result;
}

function arraysEqual(a, b) {
  for (let i = 0; i < 26; i++) {
    if (a[i] !== b[i]) return false;
  }
  return true;
}
```

// 时间 O(n)，空间 O(1)（固定 26 个字母）

```javascript
function getIntersectionNode(headA, headB) {
  if (!headA || !headB) return null;

  let pA = headA;
  let pB = headB;

  while (pA !== pB) {
    pA = pA ? pA.next : headB;
    pB = pB ? pB.next : headA;
```

### 15. LeetCode 200 - 岛屿数量（DFS）

```javascript
function numIslands(grid) {
  if (!grid || !grid.length) return 0;

  const rows = grid.length;
  const cols = grid[0].length;
  let count = 0;

  function dfs(r, c) {
    if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] !== "1") {
      return;
    }
    grid[r][c] = "0"; // 标记为已访问
    dfs(r - 1, c);
    dfs(r + 1, c);
    dfs(r, c - 1);
    dfs(r, c + 1);
  }

  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === "1") {
        count++;
        dfs(r, c);
      }
    }
  }

  return count;
}
```

// 时间 O(m*n)，空间 O(m*n)（递归栈）

### 16. LeetCode 994 - 腐烂的橘子（多源 BFS）

```javascript
function orangesRotting(grid) {
  const rows = grid.length;
  const cols = grid[0].length;
  let fresh = 0;
  const queue = [];

  // 统计新鲜橘子，并把所有腐烂橘子入队
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === 1) fresh++;
      if (grid[r][c] === 2) queue.push([r, c, 0]); // [r, c, 时间]
    }
  }

  if (fresh === 0) return 0;

  const directions = [
    [-1, 0],
    [1, 0],
    [0, -1],
    [0, 1],
  ];
  let maxTime = 0;

  while (queue.length > 0) {
    const [r, c, time] = queue.shift();
    maxTime = Math.max(maxTime, time);

    for (const [dr, dc] of directions) {
      const nr = r + dr;
      const nc = c + dc;
      if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] === 1) {
        grid[nr][nc] = 2;
        fresh--;
        queue.push([nr, nc, time + 1]);
      }
    }
  }

  return fresh === 0 ? maxTime : -1;
}
```

// 时间 O(m*n)，空间 O(m*n)

### 17. LeetCode 42 - 接雨水（双指针）

```javascript
function trap(height) {
  if (!height || height.length < 3) return 0;

  let left = 0;
  let right = height.length - 1;
  let leftMax = 0;
  let rightMax = 0;
  let water = 0;

  while (left < right) {
    if (height[left] <= height[right]) {
      if (height[left] >= leftMax) {
        leftMax = height[left];
      } else {
        water += leftMax - height[left];
      }
      left++;
    } else {
      if (height[right] >= rightMax) {
        rightMax = height[right];
      } else {
        water += rightMax - height[right];
      }
      right--;
    }
  }

  return water;
}
```

// 时间 O(n)，空间 O(1)

### 18. LeetCode 239 - 滑动窗口最大值（单调队列）

```javascript
function maxSlidingWindow(nums, k) {
  if (!nums || nums.length === 0) return [];
  if (k === 1) return nums;

  const result = [];
  const deque = []; // 存索引，单调递减

  for (let i = 0; i < nums.length; i++) {
    // 移除窗口外的元素
    if (deque.length > 0 && deque[0] <= i - k) {
      deque.shift();
    }

    // 移除比当前元素小的（它们不可能成为最大值）
    while (deque.length > 0 && nums[deque[deque.length - 1]] < nums[i]) {
      deque.pop();
    }

    deque.push(i);

    // 窗口形成后记录最大值
    if (i >= k - 1) {
      result.push(nums[deque[0]]);
    }
  }

  return result;
}
```

// 时间 O(n)，空间 O(k)

### 19. LeetCode 76 - 最小覆盖子串（滑动窗口）

```javascript
function minWindow(s, t) {
  if (!s || !t || s.length < t.length) return "";

  const tCount = new Map();
  for (let char of t) {
    tCount.set(char, (tCount.get(char) || 0) + 1);
  }

  let required = tCount.size;
  let formed = 0;
  const windowCounts = new Map();
  let left = 0;
  let minLen = Infinity;
  let minStart = 0;

  for (let right = 0; right < s.length; right++) {
    const char = s[right];
    windowCounts.set(char, (windowCounts.get(char) || 0) + 1);

    if (tCount.has(char) && windowCounts.get(char) === tCount.get(char)) {
      formed++;
    }

    while (left <= right && formed === required) {
      const currentLen = right - left + 1;
      if (currentLen < minLen) {
        minLen = currentLen;
        minStart = left;
      }

      const leftChar = s[left];
      windowCounts.set(leftChar, windowCounts.get(leftChar) - 1);

      if (
        tCount.has(leftChar) &&
        windowCounts.get(leftChar) < tCount.get(leftChar)
      ) {
        formed--;
      }
      left++;
    }
  }

  return minLen === Infinity ? "" : s.slice(minStart, minStart + minLen);
}
```

// 时间 O(|s| + |t|)，空间 O(字符集大小)

### 20. 手写快排（经典递归版）

```javascript
function quickSort(arr) {
  if (arr.length <= 1) return arr;

  const pivot = arr[arr.length - 1];
  const left = [];
  const right = [];

  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] < pivot) {
      left.push(arr[i]);
    } else {
      right.push(arr[i]);
    }
  }

  return [...quickSort(left), pivot, ...quickSort(right)];
}

// 原地版（面试更常见）
function quickSortInPlace(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return arr;

  const pivotIndex = partition(arr, left, right);
  quickSortInPlace(arr, left, pivotIndex - 1);
  quickSortInPlace(arr, pivotIndex + 1, right);

  return arr;
}

function partition(arr, left, right) {
  const pivot = arr[right];
  let i = left - 1;

  for (let j = left; j < right; j++) {
    if (arr[j] <= pivot) {
      i++;
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
  }
  [arr[i + 1], arr[right]] = [arr[right], arr[i + 1]];
  return i + 1;
}

// 原地快排（不额外开空间）
function quickSort(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return arr;

  // 随机化 pivot（防最坏情况，面试加分点）
  const pivotIndex = Math.floor(Math.random() * (right - left + 1)) + left;
  swap(arr, pivotIndex, right); // 把随机选的 pivot 放到最右

  const pivot = arr[right];
  let storeIndex = left; // 记录 < pivot 的区域边界

  for (let i = left; i < right; i++) {
    if (arr[i] < pivot) {
      // < pivot 的放左边
      swap(arr, storeIndex, i);
      storeIndex++;
    }
  }

  // 把 pivot 放回正确位置
  swap(arr, storeIndex, right);

  // 分治递归
  quickSort(arr, left, storeIndex - 1);
  quickSort(arr, storeIndex + 1, right);

  return arr;
}

function swap(arr, i, j) {
  [arr[i], arr[j]] = [arr[j], arr[i]];
}
```

### 归并排序 递归

```javascript
function mergeSort(arr) {
  if (arr.length <= 1) return arr;

  const mid = Math.floor(arr.length / 2);
  const left = mergeSort(arr.slice(0, mid));
  const right = mergeSort(arr.slice(mid));

  return merge(left, right);
}

function merge(left, right) {
  const result = [];
  let i = 0,
    j = 0;

  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) {
      result.push(left[i++]);
    } else {
      result.push(right[j++]);
    }
  }

  return result.concat(left.slice(i)).concat(right.slice(j));
}
```
